## 機能検証

map_tool の当該外部保存を検証する既存テストは見つからなかったため、最小一時プロジェクト `/tmp/b1331-navigation` を使用。同一 seed の村を CLI で2回生成（inline / external）、別プロセスで両シーンを読み、以下の12項目を確認する。

```sh
bin/godot.linuxbsd.editor.x86_64 --headless --path /tmp/b1331-navigation --script /home/hatch/workspace/godot-b1331/ai_loop/map_tool.gd -- city --scene res://inline.tscn --set 'preset="village"' --set 'name="Village"' --inline
bin/godot.linuxbsd.editor.x86_64 --headless --path /tmp/b1331-navigation --script /home/hatch/workspace/godot-b1331/ai_loop/map_tool.gd -- city --scene res://external.tscn --set 'preset="village"' --set 'name="Village"'
bin/godot.linuxbsd.editor.x86_64 --headless --path /tmp/b1331-navigation --script /tmp/b1331-navigation/verify.gd
```

一時 `project.godot`:

```ini
config_version=5
[application]
config/name="B-1331 NavigationMesh external save regression"
[rendering]
renderer/rendering_method="gl_compatibility"
```

`verify.gd`（再現用）:

```gdscript
extends SceneTree

var checks := 0
var failures := 0

func check(condition: bool, message: String) -> bool:
	checks += 1
	if not condition:
		failures += 1
		printerr("FAIL: " + message)
	return condition

func _initialize() -> void:
	var inline_scene := load("res://inline.tscn") as PackedScene
	var external_scene := load("res://external.tscn") as PackedScene
	if not check(inline_scene != null and external_scene != null, "both scenes load"):
		quit(1)
		return
	var inline_root := inline_scene.instantiate()
	var external_root := external_scene.instantiate()
	var original := inline_root.get_node_or_null("Village/Navigation") as NavigationRegion3D
	var saved := external_root.get_node_or_null("Village/Navigation") as NavigationRegion3D
	if not check(original != null and saved != null, "both NavigationRegion3D nodes load"):
		inline_root.free()
		external_root.free()
		quit(1)
		return
	var before := original.navigation_mesh
	var after := saved.navigation_mesh
	if not check(before != null and after != null, "both NavigationMesh resources load"):
		inline_root.free()
		external_root.free()
		quit(1)
		return
	check(before.get_polygon_count() > 0 and before.get_vertices().size() > 0, "navigation was baked")
	var expected_path := "res://external_maps/Village/Navigation.res"
	check(after.resource_path == expected_path, "NavigationMesh uses a separate resource path")
	check(FileAccess.file_exists(expected_path), "NavigationMesh file exists")
	var scene_text := FileAccess.get_file_as_string("res://external.tscn")
	var declares_external := false
	for line in scene_text.split("\n"):
		declares_external = declares_external or (line.begins_with("[ext_resource ") and line.contains('type="NavigationMesh"') and line.contains('path="' + expected_path + '"'))
	check(declares_external, "scene declares an external NavigationMesh")
	check(not scene_text.contains('[sub_resource type="NavigationMesh"'), "scene contains no inline NavigationMesh")
	var separate := ResourceLoader.load(expected_path, "NavigationMesh", ResourceLoader.CACHE_MODE_IGNORE) as NavigationMesh
	check(separate != null, "NavigationMesh file loads independently")
	check(after.get_vertices() == before.get_vertices(), "baked vertices survive reload")
	check(after.get_polygon_count() == before.get_polygon_count(), "polygon count survives reload")
	var polygons_equal := true
	for index in before.get_polygon_count():
		polygons_equal = polygons_equal and before.get_polygon(index) == after.get_polygon(index)
	check(polygons_equal, "all polygon indices survive reload")
	print("B1331_RESULT " + JSON.stringify({"ok": failures == 0, "checks": checks, "vertices": after.get_vertices().size(), "polygons": after.get_polygon_count(), "resource": after.resource_path}))
	inline_root.free()
	external_root.free()
	quit(0 if failures == 0 else 1)
```

12 項目すべて pass（終了コード 0）。別プロセスで `res://external_maps/Village/Navigation.res` をシーン経由と単独の両方で読み込み、頂点 3,588 個、ポリゴン 3,795 個と全ポリゴンの頂点インデックスが inline 保存と一致した。`.tscn` は `ext_resource` を参照し、`NavigationMesh` の `sub_resource` は 0 件。外部ファイルは 45 個で、`Navigation.res` は 32,335 バイト。

比較として、同じエディタ・同じ村を変更前の `map_tool.gd` で外部保存した。外部ファイルは 44 個で `Navigation.res` は生成されず、同じ検証で外部保存に関する 5 項目が fail（終了コード 1）。
